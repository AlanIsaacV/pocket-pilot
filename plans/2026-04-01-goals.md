```go
package goal

import "fmt"

func ValidateGoal(g Goal) error {
    if normalizeName(g.Name) == "" {
        return fmt.Errorf("name is required")
    }
    if g.TargetAmountCents <= 0 {
        return fmt.Errorf("target_amount_cents must be > 0")
    }
    if g.SavedAmountCents < 0 {
        return fmt.Errorf("saved_amount_cents must be >= 0")
    }
    if g.SavedAmountCents > g.TargetAmountCents {
        return fmt.Errorf("saved_amount_cents must be <= target_amount_cents")
    }
    if g.TargetDate.IsZero() {
        return fmt.Errorf("target_date is required")
    }
    return nil
}
```

```dart
// frontend/lib/src/features/goals/domain/goal.dart
import 'package:fixnum/fixnum.dart';
import 'package:pocket_pilot/src/generated/pocket_pilot/v1/goal_service.pbgrpc.dart';

class GoalModel {
  final String id;
  final String name;
  final int targetAmountCents;
  final int savedAmountCents;
  final String targetDate;
  final String? categoryId;
  final DateTime createdAt;
  final DateTime updatedAt;
  final DateTime? deletedAt;

  const GoalModel({
    required this.id,
    required this.name,
    required this.targetAmountCents,
    required this.savedAmountCents,
    required this.targetDate,
    this.categoryId,
    required this.createdAt,
    required this.updatedAt,
    this.deletedAt,
  });

  factory GoalModel.fromProto(Goal proto) {
    return GoalModel(
      id: proto.id,
      name: proto.name,
      targetAmountCents: proto.targetAmountCents.toInt(),
      savedAmountCents: proto.savedAmountCents.toInt(),
      targetDate: _dateOnly(proto.targetDate),
      categoryId: proto.hasCategoryId() ? proto.categoryId : null,
      createdAt: proto.createdAt.toDateTime().toUtc(),
      updatedAt: proto.updatedAt.toDateTime().toUtc(),
      deletedAt: proto.hasDeletedAt() ? proto.deletedAt.toDateTime().toUtc() : null,
    );
  }

  factory GoalModel.fromMap(Map<String, dynamic> map) {
    return GoalModel(
      id: map['id'] as String,
      name: map['name'] as String,
      targetAmountCents: _asInt(map['target_amount_cents']),
      savedAmountCents: _asInt(map['saved_amount_cents']),
      targetDate: _dateOnly(map['target_date'] as String),
      categoryId: map['category_id'] as String?,
      createdAt: DateTime.parse(map['created_at'] as String).toUtc(),
      updatedAt: DateTime.parse(map['updated_at'] as String).toUtc(),
      deletedAt: map['deleted_at'] != null
          ? DateTime.parse(map['deleted_at'] as String).toUtc()
          : null,
    );
  }

  Map<String, dynamic> toMap() {
    return {
      'id': id,
      'name': name.trim(),
      'target_amount_cents': targetAmountCents,
      'saved_amount_cents': savedAmountCents,
      'target_date': _dateOnly(targetDate),
      'category_id': categoryId,
      'created_at': createdAt.toIso8601String(),
      'updated_at': updatedAt.toIso8601String(),
      'deleted_at': deletedAt?.toIso8601String(),
    };
  }

  Map<String, dynamic> toServerMap() => toMap();

  Goal toProto() {
    return Goal(
      id: id,
      name: name.trim(),
      targetAmountCents: Int64(targetAmountCents),
      savedAmountCents: Int64(savedAmountCents),
      targetDate: _dateOnly(targetDate),
      categoryId: categoryId,
    );
  }

  double get progressPercentage {
    if (targetAmountCents <= 0) return 0;
    return (savedAmountCents / targetAmountCents) * 100;
  }

  int monthsRemainingAt(DateTime nowLocal) {
    final now = DateTime(nowLocal.year, nowLocal.month, 1);
    final target = DateTime.parse(_dateOnly(targetDate));
    final raw = (target.year - now.year) * 12 + (target.month - now.month);
    return raw <= 0 ? 1 : raw;
  }

  int get monthsRemaining => monthsRemainingAt(DateTime.now());

  int monthlyRequiredCentsAt(DateTime nowLocal) {
    final remaining = targetAmountCents - savedAmountCents;
    if (remaining <= 0) return 0;
    final months = monthsRemainingAt(nowLocal);
    return (remaining + months - 1) ~/ months;
  }

  int get monthlyRequiredCents => monthlyRequiredCentsAt(DateTime.now());

  bool isOverdueAt(DateTime nowLocal) {
    if (savedAmountCents >= targetAmountCents) return false;
    final now = DateTime(nowLocal.year, nowLocal.month, 1);
    final target = DateTime.parse(_dateOnly(targetDate));
    final targetMonth = DateTime(target.year, target.month, 1);
    return targetMonth.isBefore(now);
  }

  bool get isOverdue => isOverdueAt(DateTime.now());

  static String _dateOnly(String value) {
    return value.length >= 10 ? value.substring(0, 10) : value;
  }

  static int _asInt(Object raw) {
    if (raw is int) return raw;
    if (raw is num) return raw.toInt();
    return int.parse(raw.toString());
  }
}

class GoalMath {
  static int totalMonthlyRequiredCents(
    Iterable<GoalModel> goals, {
    DateTime? now,
  }) {
    final current = now ?? DateTime.now();
    return goals.where((g) => g.deletedAt == null).fold(0, (sum, g) {
      return sum + g.monthlyRequiredCentsAt(current);
    });
  }
}
```

```dart
import 'package:sqflite/sqflite.dart';
import 'package:pocket_pilot/src/core/db/database_helper.dart';

class GoalLocalStore {
  final DatabaseHelper _db;

  GoalLocalStore({required DatabaseHelper db}) : _db = db;

  Future<List<Map<String, dynamic>>> queryRows() async {
    final db = await _db.database;
    return db.query(
      'goals',
      where: 'deleted_at IS NULL',
      orderBy: 'created_at DESC',
    );
  }

  Future<void> upsertRow(Map<String, dynamic> row) async {
    final db = await _db.database;
    await db.insert('goals', row, conflictAlgorithm: ConflictAlgorithm.replace);
  }

  Future<void> upsertRows(List<Map<String, dynamic>> rows) async {
    final db = await _db.database;
    final batch = db.batch();
    for (final row in rows) {
      batch.insert('goals', row, conflictAlgorithm: ConflictAlgorithm.replace);
    }
    await batch.commit(noResult: true);
  }

  Future<void> softDelete(String id, DateTime deletedAt) async {
    final db = await _db.database;
    final ts = deletedAt.toUtc().toIso8601String();
    await db.update(
      'goals',
      {'deleted_at': ts, 'updated_at': ts},
      where: 'id = ? AND (deleted_at IS NULL OR deleted_at < ?)',
      whereArgs: [id, ts],
    );
  }
}
```

```dart
import 'dart:convert';

import 'package:pocket_pilot/src/core/sync/sync_outbox.dart';
import 'package:pocket_pilot/src/features/goals/data/goal_local_store.dart';
import 'package:pocket_pilot/src/features/goals/domain/goal.dart';

class GoalRepository {
  final GoalLocalStore _local;
  final SyncOutbox _outbox;

  GoalRepository({required GoalLocalStore local, required SyncOutbox outbox})
      : _local = local,
        _outbox = outbox;

  Future<List<GoalModel>> getGoals() async {
    final rows = await _local.queryRows();
    return rows.map((row) => GoalModel.fromMap(row)).toList();
  }

  Future<void> upsertGoal(GoalModel goal) async {
    await _local.upsertRow(goal.toMap());
    await _outbox.enqueue(
      entityType: 'goal',
      entityId: goal.id,
      operation: 'upsert',
      payload: jsonEncode(goal.toServerMap()),
    );
  }

  Future<void> deleteGoal(String id) async {
    final now = DateTime.now().toUtc();
    await _local.softDelete(id, now);
    await _outbox.enqueue(
      entityType: 'goal',
      entityId: id,
      operation: 'delete',
      payload: jsonEncode({'deleted_at': now.toIso8601String()}),
    );
  }
}
```

```dart
import 'dart:async';

import 'package:flutter/foundation.dart';
import 'package:pocket_pilot/src/core/sync/sync_service.dart';
import 'package:pocket_pilot/src/features/goals/data/goal_repository.dart';
import 'package:pocket_pilot/src/features/goals/domain/goal.dart';

class GoalViewModel extends ChangeNotifier {
  final GoalRepository _repository;
  final SyncService _syncService;

  List<GoalModel> goals = [];
  bool isLoading = false;
  bool isSyncing = false;
  String? errorMessage;
  DateTime? _lastSyncAt;

  GoalViewModel({
    required GoalRepository repository,
    required SyncService syncService,
  })  : _repository = repository,
        _syncService = syncService;

  int get totalMonthlyRequiredCents => GoalMath.totalMonthlyRequiredCents(goals);

  List<GoalModel> get overdueGoals => goals.where((g) => g.deletedAt == null && g.isOverdue).toList(growable: false);

  void loadOnAppear() {
    unawaited(_loadLocalThenMaybeSync());
  }

  Future<void> refresh() => _runSync();

  Future<void> upsertGoal(GoalModel goal) async {
    await _repository.upsertGoal(goal);
    await _reloadLocal();
  }

  Future<void> deleteGoal(String id) async {
    await _repository.deleteGoal(id);
    await _reloadLocal();
  }

  Future<void> _reloadLocal() async {
    goals = await _repository.getGoals();
    notifyListeners();
  }

  Future<void> _loadLocalThenMaybeSync() async {
    await _reloadLocal();
    await _maybeSyncRemote();
  }

  Future<void> _maybeSyncRemote() async {
    final last = _lastSyncAt;
    if (last != null && DateTime.now().difference(last) < const Duration(seconds: 5)) {
      return;
    }
    await _runSync();
  }

  Future<void> _runSync() async {
    isLoading = goals.isEmpty;
    isSyncing = true;
    notifyListeners();

    try {
      await _syncService.sync();
      goals = await _repository.getGoals();
      _lastSyncAt = DateTime.now();
      errorMessage = null;
    } catch (e) {
      errorMessage = e.toString();
    } finally {
      isLoading = false;
      isSyncing = false;
      notifyListeners();
    }
  }
}
```
